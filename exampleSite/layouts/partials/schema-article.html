Questo partial colma quel buco. Incluso da head.html. Si attiva solo su .IsPage nelle sezioni blog-like, quindi non tocca hub destinazione (schema proprio), pagine statiche o liste. Il dict viene costruito e poi passato a jsonify (come in layouts/destinazioni/single.html) per evitare problemi di escaping a mano. */ -}} {{- if and .IsPage (in (slice "blog" "post" "posts" "news") .Section) -}} {{- /* Immagine: Cloudinary (URL http assoluto) usata così com'è; immagine in page bundle o sotto static/ risolta ad URL assoluto; fallback all'og image del sito. Se resta vuota, la chiave "image" viene omessa. */ -}} {{- $img := "" -}} {{- with .Params.image -}} {{- if hasPrefix . "http" -}} {{- $img = . -}} {{- else -}} {{- with $.Resources.GetMatch . -}} {{- $img = .Permalink -}} {{- else -}} {{- if fileExists (printf "static/%s" .) -}} {{- $img = (. | absURL) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $img -}} {{- with site.Params.image -}} {{- if not (hasSuffix (lower .) ".svg") -}} {{- $img = (. | absURL) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Logo publisher: risorsa in assets/, va pubblicata e linkata come URL assoluto (un semplice "logo | absURL" punterebbe a un file inesistente sotto static/). .RelPermalink | absLangURL tiene l'URL sul dominio della lingua corrente in modalità multihost (.Permalink della risorsa restituirebbe sempre vandipety.it anche nelle pagine EN). */ -}} {{- $logoURL := "" -}} {{- with resources.Get site.Params.logo -}} {{- $logoURL = .RelPermalink | absLangURL -}} {{- end -}} {{- $sameAs := slice -}} {{- range site.Params.social -}} {{- with .link -}}{{- $sameAs = $sameAs | append . -}}{{- end -}} {{- end -}} {{- $desc := cond (ne (string .Description) "") (string .Description) (.Summary | plainify) -}} {{- $author := .Params.author | default site.Params.author -}} {{- $isoFmt := "2006-01-02T15:04:05Z07:00" -}} {{- $publisher := dict "@type" "Organization" "name" site.Title "url" site.BaseURL -}} {{- with $logoURL -}} {{- $publisher = merge $publisher (dict "logo" (dict "@type" "ImageObject" "url" .)) -}} {{- end -}} {{- with $sameAs -}} {{- $publisher = merge $publisher (dict "sameAs" .) -}} {{- end -}} {{- $schema := dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" (.Title | plainify | truncate 110 "") "description" $desc "url" .Permalink "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "datePublished" (.PublishDate.Format $isoFmt) "dateModified" (.Lastmod.Format $isoFmt) "inLanguage" site.Language.Lang "author" (dict "@type" "Person" "name" $author "url" (absLangURL "about/")) "publisher" $publisher -}} {{- with $img -}} {{- $schema = merge $schema (dict "image" (slice .)) -}} {{- end -}} {{- with .Params.categories -}} {{- $schema = merge $schema (dict "articleSection" (index . 0) "keywords" (delimit . ", ")) -}} {{- end -}} {{- /* BreadcrumbList: Home › Categoria › Articolo.
