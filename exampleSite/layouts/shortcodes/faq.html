{{- /* Blocco FAQ + dati strutturati FAQPage (schema.org) -> possibile rich snippet. Uso nel markdown: ogni domanda è una riga che inizia con "### ", la risposta è il testo (markdown) fino alla domanda successiva. {{< faq >}} ### Quanto costa vivere in camper? Dipende dallo stile di viaggio. Nel nostro caso la media è di circa **X € al giorno**. ### Serve la patente internazionale? Per l'Europa no; fuori dall'area della Convenzione di Vienna sì. {{< /faq >}} */ -}} {{- $chunks := split (printf "\n%s" (trim .Inner "\n \t\r")) "\n### " -}} {{- $items := slice -}}
{{- range $i, $c := $chunks -}} {{- $c = trim $c "\n \t\r" -}} {{- if and (gt $i 0) $c -}} {{- $parts := split $c "\n" -}} {{- $q := trim (index $parts 0) " #" -}} {{- $a := trim (delimit (after 1 $parts) "\n") "\n \t\r" -}} {{- $aHTML := $.Page.RenderString $a -}} {{- $items = $items | append (dict "q" $q "a" (string $aHTML)) -}}
{{ $q }}
{{ $aHTML }}
{{- end -}} {{- end -}}
